#include <stdio.h>

int main() {

	char gender;
	int age;
	double height;

	getchar(); // 엔터 값도 흡수


	printf("나이는? >> ");

	getchar();

	printf("키는? >>");
	scanf("%lf", &height);

	printf("================\n");
	printf("성별 : %c\n", gender);
	printf("나이 : %3d\n", age);
	printf("키 : %.lf\n", height);


	return 0;
}

## 실습 7번
```c
#include <stdio.h>

int main()
{
    int age;
    char gender;
    double height;
    
    // 자료 입력 받기
    printf("성별은? (남자라면 M, 여자라면 F) \n");
    scanf(" %c", &gender); // 공백 추가하여 개행 문자 무시
    printf("나이는? \n");
    scanf("%d", &age); // 정수 입력
    printf("키는? \n");
    scanf("%lf", &height); // 실수 입력
    
    printf("\n=====================\n");
    printf("성별: %c\n", gender);
    printf("나이: %d\n", age);
    printf("키: %.2lf\n", height); // 키는 소수점 둘째 자리까지 출력
    
    return 0;
}

```

## 실습 10-1
```c
#include <stdio.h>

int main() {
    const double pi = 3.141592;
    int num;
    int r;

    do {
        printf("==================\n");
        printf("1. 원의 둘레 구하기\n");
        printf("2. 원의 넓이 구하기\n");
        printf("3. 구의 부피 구하기\n");
        printf("4. 그만두기\n");
        printf("==================\n");
        printf("원하는 번호는: ");

        scanf_s("%d", &num);
        getchar();  // 버퍼 비우기
        
        if (num == 4) break;
        else if (num > 4 || num < 1) {
            printf("잘못된 번호입니다\n");
        }
        else {
            printf("반지름은?: ");
            scanf_s("%d", &r);
            getchar();  // 버퍼 비우기

            switch (num) {
            case 1:
                printf("반지름이 %d인 원의 둘레는: %.2lf\n", r, r * 2 * pi);
                break;
            case 2:
                printf("반지름이 %d인 원의 넓이는: %.2lf\n", r, r * r * pi);
                break;
            case 3:
                printf("반지름이 %d인 구의 부피는: %.2lf\n", r, (4.0 / 3.0) * pi * r * r * r);
                break;
            default:
                break;
            }
        }
    } while (1);
    return 0;
}

```
