#define _CRT_SECURE_NO_WARNINGS

#include <curses.h>
#include <string.h>

#define NUM_MENU_ITEMS 4
#define MAX_POSTS 10
#define MAX_TITLE_LEN 50
#define MAX_CONTENT_LEN 200

const char* menu_items[NUM_MENU_ITEMS] = {
    "1. Write Post",
    "2. View Posts",
    "3. Settings",
    "4. ESC"
};

typedef struct {
    int id;
    char title[MAX_TITLE_LEN];
    char content[MAX_CONTENT_LEN];
} Post;

Post posts[MAX_POSTS];
int num_posts = 0;

void display_post_content(int post_index) {
    clear();
    mvprintw(1, 5, "==== Post Content ====");
    mvprintw(3, 5, "[Title] \n");
    mvprintw(4, 5, "%s", posts[post_index].title);
    mvprintw(6, 5, "[Content] \n");
    mvprintw(7, 5, "%s", posts[post_index].content);
    mvprintw(9, 5, "Press LEFT arrow key to go back.");
    refresh();
    while (1) {
        int ch = getch();
        if (ch == KEY_LEFT) {
            return;  // Go back to the previous screen
        }
    }
}

void display_posts() {
    int highlight = 0;  // ���� ���õ� �Խñ� �ε���

    while (1) {
        clear();
        mvprintw(1, 5, "==== Posts ====");
        if (num_posts == 0) {
            mvprintw(3, 5, "No posts available.");
        }
        else {
            // �Խñ� ���� ���
            for (int i = 0; i < num_posts; i++) {
                if (i == highlight) {
                    attron(A_REVERSE); // ���� ȿ��
                    mvprintw(3 + i, 5, "%d. %s", i + 1, posts[i].title);
                    attroff(A_REVERSE);
                }
                else {
                    mvprintw(3 + i, 5, "%d. %s", i + 1, posts[i].title);
                }
            }
        }
        mvprintw(3 + num_posts + 1, 5, "Use UP/DOWN arrow keys to select a post, ENTER to view content, LEFT arrow to go back.");
        refresh();

        int ch = getch();
        switch (ch) {
        case KEY_UP: // �� Ű: ���� �̵�
            highlight = (highlight > 0) ? highlight - 1 : num_posts - 1;
            break;
        case KEY_DOWN: // �� Ű: �Ʒ��� �̵�
            highlight = (highlight < num_posts - 1) ? highlight + 1 : 0;
            break;
        case '\n': // Enter Ű: ���õ� �Խñ��� ���� ����
            display_post_content(highlight);
            break;
        case KEY_LEFT: // ���� ȭ��ǥ: ���ư���
            return;
        }
    }
}

void write_post() {
    clear();
    mvprintw(1, 5, "==== Write Post ====");
    char title[MAX_TITLE_LEN];
    char content[MAX_CONTENT_LEN];

    mvprintw(3, 5, "Enter title: ");
    echo();
    getstr(title);
    noecho();

    mvprintw(5, 5, "Enter content: ");
    echo();
    getstr(content);
    noecho();

    if (num_posts < MAX_POSTS) {
        strncpy(posts[num_posts].title, title, MAX_TITLE_LEN);
        strncpy(posts[num_posts].content, content, MAX_CONTENT_LEN);
        num_posts++;
    }
    else {
        mvprintw(7, 5, "Post limit reached. Cannot add more posts.");
    }

    mvprintw(9, 5, "Press LEFT arrow key to go back.");
    refresh();
    while (1) {
        int ch = getch();
        if (ch == KEY_LEFT) {
            return;  // Go back to the previous screen
        }
    }
}

int main() {
    initscr();            // PDCurses �ʱ�ȭ
    noecho();             // �Է� ���� ����
    cbreak();             // �� ���۸� ��Ȱ��ȭ
    keypad(stdscr, TRUE); // ����Ű Ȱ��ȭ

    int highlight = 0;    // ���� ���õ� �޴� �ε���
    int choice = -1;      // ���õ� �޴� (�ʱⰪ -1)

    while (1) {
        clear();  // ȭ�� �ʱ�ȭ

        mvprintw(1, 5, "==== Menu ===="); // ���� ���

        // �޴� �׸� ��� (���õ� �׸��� ����)
        for (int i = 0; i < NUM_MENU_ITEMS; i++) {
            if (i == highlight) {
                attron(A_REVERSE); // ���� ȿ��
                mvprintw(3 + i, 5, menu_items[i]);
                attroff(A_REVERSE);
            }
            else {
                mvprintw(3 + i, 5, menu_items[i]);
            }
        }

        mvprintw(3 + NUM_MENU_ITEMS, 5, "Use UP/DOWN arrow keys to navigate, LEFT arrow to go back.");
        refresh();

        int ch = getch();
        switch (ch) {
        case KEY_UP: // �� Ű: ���� �̵�
            highlight = (highlight > 0) ? highlight - 1 : NUM_MENU_ITEMS - 1;
            break;
        case KEY_DOWN: // �� Ű: �Ʒ��� �̵�
            highlight = (highlight < NUM_MENU_ITEMS - 1) ? highlight + 1 : 0;
            break;
        case '\n': // Enter Ű: ����
            choice = highlight;
            break;
        case KEY_LEFT: // ���� ȭ��ǥ: ���� �Ǵ� �ڷΰ���
            endwin();
            return 0;
        }

        // ���õ� �޴� ����
        if (choice != -1) {
            clear();
            switch (choice) {
            case 0: // Write Post
                write_post();
                break;
            case 1: // View Posts
                display_posts();
                break;
            case 2: // Settings (��� �߰� ����)
                mvprintw(10, 5, "Settings - Not implemented yet.");
                refresh();
                getch();
                break;
            case 3: // ESC
                endwin();
                return 0;
            }
            choice = -1; // ���� �ʱ�ȭ
        }

        refresh();
    }

    endwin();
    return 0;
}
